function importPIBNominalValueData() {
  // Token - API de Banxico
  var token = "5cec8fbf16c0d5aaad4816e1a6069839d82d5962de8a5899577d1b728fb887b2";
  
  // Date range: January 1993 to present
  var startDate = "1993-01-01";
  var endDate = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), "yyyy-MM-dd");
  
  // URL for data serie item - PIB (SR17645) - Nominal Value Quarterly
  var baseUrl = "https://www.banxico.org.mx/SieAPIRest/service/v1";
  var url = baseUrl + "/series/SR17645/datos/" + startDate + "/" + endDate;
  
  // Options for HTTP request
  var options = {
    "headers": {
      "Bmx-Token": token,
      "Accept": "application/json"
    }
  };
  
  Logger.log("Requesting data from Banxico API...");
  Logger.log("URL: " + url);
  
  try {
    // API request
    var response = UrlFetchApp.fetch(url, options);
    var json = JSON.parse(response.getContentText());
    var seriesData = json.bmx.series[0].datos;
    
    Logger.log("Raw data received: " + seriesData.length + " records");
    
    // Log a few sample records to debug
    for (var i = 0; i < Math.min(3, seriesData.length); i++) {
      Logger.log("Sample record " + i + ": " + JSON.stringify(seriesData[i]));
    }
    
    // Data processing
    var data = [];
    var errorCount = 0;
    
    seriesData.forEach(function(item, index) {
      try {
        var dateStr = item.fecha;
        var dateParts = dateStr.split('/');
        
        if (dateParts.length !== 3) return;
        
        var startMonth = parseInt(dateParts[1], 10) - 1; // JavaScript months are 0-indexed
        var year = parseInt(dateParts[2], 10);
        
        // Map start of quarter to end of quarter month and quarter string
        var quarter = "";
        var endMonth = startMonth;
        
        if (startMonth === 0) {       // January -> Quarter ends March 31
          quarter = "Q1";
          endMonth = 2;
        } else if (startMonth === 3) { // April -> Quarter ends June 30
          quarter = "Q2";
          endMonth = 5;
        } else if (startMonth === 6) { // July -> Quarter ends September 30
          quarter = "Q3";
          endMonth = 8;
        } else if (startMonth === 9) { // October -> Quarter ends December 31
          quarter = "Q4";
          endMonth = 11;
        }

        // Setting day parameter to 0 in Date constructor gets the last day of 'endMonth'
        var quarterEndDate = new Date(year, endMonth + 1, 0);

        // Process value: strip commas to allow correct parseFloat
        var value = null;
        if (item.dato && item.dato !== "N/E") {
          var cleanString = item.dato.replace(/,/g, '');
          var parsedValue = parseFloat(cleanString);
          
          if (!isNaN(parsedValue)) {
            value = parsedValue;
          }
        }

        data.push({
          date: quarterEndDate,
          quarter: quarter,
          value: value
        });
        
      } catch (e) {
        Logger.log("Error processing record " + index + ": " + e.toString());
        errorCount++;
      }
    });

    Logger.log("Processed " + data.length + " valid records, " + errorCount + " errors");
    
    if (data.length === 0) {
      throw new Error("No valid data to process");
    }
    
    // Sort by date (ascending)
    data.sort(function(a, b) {
      return a.date - b.date;
    });
    
    Logger.log("Date range: from " + data[0].date.toLocaleDateString() + " to " + data[data.length - 1].date.toLocaleDateString());
    
    // Get or create spreadsheet
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheetName = "Data PIB Nominal Value";
    var sheet = ss.getSheetByName(sheetName);
    
    if (sheet) {
      sheet.clear();
    } else {
      sheet = ss.insertSheet(sheetName);
    }
    
    // Build 2D array for batch writing with 3 columns
    var sheetData = [];
    
    // Add headers
    sheetData.push(["Date (DD/MM/YYYY)", "Quarter", "PIB - Nominal Value in Millions MXN"]);
    
    // Add data rows
    data.forEach(function(item) {
      sheetData.push([item.date, item.quarter, item.value]);
    });
    
    Logger.log("Preparing to write " + sheetData.length + " rows to Spreadsheet");
    Logger.log("First data row: " + JSON.stringify(sheetData[1]));
    Logger.log("Last data row: " + JSON.stringify(sheetData[sheetData.length - 1]));
    
    // Write data to spreadsheet
    if (sheetData.length > 1) {
      // Clear existing content
      sheet.getRange(1, 1, sheet.getMaxRows(), sheet.getMaxColumns()).clearContent();
      
      // Write data across 3 columns
      var range = sheet.getRange(1, 1, sheetData.length, 3);
      range.setValues(sheetData);
      
      // Column 1: Date formatting (dd/MM/yyyy)
      var dateRange = sheet.getRange(2, 1, sheetData.length - 1, 1);
      dateRange.setNumberFormat("dd/MM/yyyy");
      
      // Column 2: Center align Quarter values
      var quarterRange = sheet.getRange(2, 2, sheetData.length - 1, 1);
      quarterRange.setHorizontalAlignment("center");
      
      // Column 3: Numeric formatting with thousands separators
      var valueRange = sheet.getRange(2, 3, sheetData.length - 1, 1);
      valueRange.setNumberFormat("#,##0.00");
      
      // Auto-resize columns
      sheet.autoResizeColumns(1, 3);
      
      Logger.log("Successfully wrote " + (sheetData.length - 1) + " data rows to spreadsheet");
      
    } else {
      Logger.log("No data rows to write");
    }
    
  } catch (e) {
    Logger.log("Import Error: " + e.toString());
    Logger.log("Stack: " + e.stack);
    
    try {
      SpreadsheetApp.getUi().alert(
        'Import Error',
        'Failed to import data: ' + e.toString(),
        SpreadsheetApp.getUi().ButtonSet.OK
      );
    } catch (uiError) {
      Logger.log("Could not show UI alert: " + uiError.toString());
    }
  }
}
