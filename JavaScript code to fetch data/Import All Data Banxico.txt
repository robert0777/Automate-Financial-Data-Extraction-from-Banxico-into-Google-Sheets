// Master function to execute all imports
function importAllData() {
  Logger.log("Starting import of all Banxico data...");
  
  try {
    importCETES28Data();
    Logger.log("✓ CETES 28-days data imported successfully");


    importCETES91Data();
    Logger.log("✓ CETES 91-days data imported successfully");



    importCETES182Data();
    Logger.log("✓ CETES 182-days data imported successfully");

    importTIIP28Data();
    Logger.log("✓ TIIP 28-days data imported successfully");



    importTIIE28Data();
    Logger.log("✓ TIIE 28-days data imported successfully");


    importBONDES3YData();
    Logger.log("✓ BONDES 3-years data imported successfully");


    importUDIBONOS3YData();
    Logger.log("✓ UDIBONOS 3-years data imported successfully");






    
    importCPPData();
    Logger.log("✓ CPP data imported successfully");
    

    
    importINPCData();
    Logger.log("✓ INPC data imported successfully");
    
    importSalarioMinimoData();
    Logger.log("✓ Salario Minimo data imported successfully");
    



    importExchangeRateMXN_USDData();
    Logger.log("✓ Exchange Rate data imported successfully");




    importUDIsData();
    Logger.log("✓ UDIs data imported successfully");

    importIPC_BMVData();
    Logger.log("✓ IPC BMV data imported successfully");


    importCreditoHogaresData();
    Logger.log("✓ Tasa de Crédito a los Hogares data imported successfully");



    Logger.log("All data imports completed successfully!");
    
    // Optional: Show completion message to user
    SpreadsheetApp.getUi().alert('Success', 'All Banxico data has been imported successfully!', SpreadsheetApp.getUi().ButtonSet.OK);
    
  } catch (error) {
    Logger.log("Error during batch import: " + error.toString());
    SpreadsheetApp.getUi().alert('Error', 'An error occurred during import: ' + error.toString(), SpreadsheetApp.getUi().ButtonSet.OK);
  }
}
